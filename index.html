function processWhatsAppOrder() { const name = document.getElementById('custName').value.trim(); let phone = document.getElementById('custPhone').value.trim(); const address = document.getElementById('custAddress').value.trim(); const city = document.getElementById('custCity').value.trim(); const pincode = document.getElementById('custPincode').value.trim(); if (!name || !phone || !address || !city || !pincode) { alert("Kripya apna Naam, Phone Number, Poora Address, City aur Pincode zaroor bharein!"); return; } const keys = Object.keys(state.cart); if(!keys.length) { alert("Aapka basket khali hai!"); return; } phone = phone.replace(/^0+/, ''); let orderSummary = ""; let itemsArray = []; keys.forEach((k, idx) => { const it = state.cart[k]; orderSummary += `${idx + 1}. ${it.name} [${it.size}] x ${it.qty} = Rs. ${it.price * it.qty}\n`; itemsArray.push(`${it.name} (${it.size}) x${it.qty}`); }); const subtotal = document.getElementById('summarySubtotal').textContent; const weight = document.getElementById('summaryWeight').textContent; const shipping = document.getElementById('summaryShipping').textContent; const grandTotal = document.getElementById('summaryGrandTotal').textContent; const upi = state.settings.UPI_ID || '9690002789@upi'; const waNumber = state.settings.WhatsAppNumber || '919690002789'; // Reliable Background Write to Google Sheet const payload = new URLSearchParams(); payload.append("name", name); payload.append("phone", phone); payload.append("address", address); payload.append("city", city); payload.append("pincode", pincode); payload.append("items", itemsArray.join(", ")); payload.append("weight", weight); payload.append("grandTotal", grandTotal); fetch(SHEET_URLS.orderScript, { method: "POST", mode: "no-cors", headers: { "Content-Type": "application/x-www-form-urlencoded" }, body: payload.toString() }); const fullMessage = `*NEW ORDER - SHREEJU FOODS*\n\n` + `*CUSTOMER DETAILS:*\n` + `Name: ${name}\n` + `Phone: ${phone}\n` + `Address: ${address}\n` + `City: ${city}\n` + `Pincode: ${pincode}\n\n` + `*ITEMS ORDERED:*\n` + orderSummary + `\n*BILL SUMMARY:*\n` + `Subtotal: ${subtotal}\n` + `Weight: ${weight}\n` + `Shipping: ${shipping}\n` + `*TOTAL PAYABLE:* ${grandTotal}\n\n` + `Payment: Store UPI (${upi}) / Confirmation par baat karein`; const waURL = `https://wa.me/${waNumber}?text=${encodeURIComponent(fullMessage)}`; window.open(waURL, '_blank'); }
